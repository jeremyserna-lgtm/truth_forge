### Phase 2: Build Cross-Layer Intelligence

**Create Integration Tools** in `src/truth_forge/mcp/integration/`:

```python
# cognitive_stack.py

class CognitiveStack:
    """Orchestrates MCP servers as unified cognitive system."""

    def __init__(self):
        self.perception = PerceptionLayer()
        self.memory = MemoryLayer()
        self.cognition = CognitionLayer()
        self.knowledge = KnowledgeLayer()
        self.action = ActionLayer()

    async def process_query(self, query: str) -> dict:
        """Process query through complete cognitive stack.

        HOLD₁: Query
        AGENT: Multi-layer processing
        HOLD₂: Integrated result
        """
        # 1. PERCEPTION: Understand what's being asked
        intent = await self.perception.parse_intent(query)

        # 2. MEMORY: Retrieve relevant context
        context = await self.memory.recall(
            query=query,
            intent=intent,
            sources=["bigquery", "knowledge_graph", "spine_analysis"]
        )

        # 3. COGNITION: Reason about the query
        reasoning = await self.cognition.think(
            query=query,
            context=context,
            method="sequential_thinking"
        )

        # 4. KNOWLEDGE: Integrate insights
        knowledge = await self.knowledge.synthesize(
            reasoning=reasoning,
            context=context
        )

        # 5. ACTION: Execute if needed
        if reasoning.requires_action:
            result = await self.action.execute(
                plan=reasoning.plan,
                governance_check=True
            )
        else:
            result = knowledge

        return result
```
